# M3 Tool Integration Implementation Plan

> **For agentic workers:** In this plan, the code tasks for `src/employee-api` and `src/mcp-server` are done directly by AI (write code, run tests, commit); the first Dev Tunnels login and the Copilot Studio / Power Apps maker work are done by the user in the browser/terminal, and AI updates the notes and coverage table and commits at the end of each Task. Steps are tracked with checkboxes (`- [ ]`).

**Goal:** On `HR Assistant Classic`, connect the same employee data in four different ways (direct OpenAPI, custom connector, Send HTTP request node, MCP), to fully cover the Add tools to agents group of exam objectives plus the API/Send HTTP request bullet in Configure topics, and along the way switch M1's "leave balance" topic from querying Dataverse directly to calling a tool.

**Architecture:** Two new independent small services: `src/employee-api` (Node.js built-in `http` module, reads data from `data/employees.csv`, 3 endpoints, API key check, with an OpenAPI 3.0 definition) and `src/mcp-server` (official `@modelcontextprotocol/sdk`, Streamable HTTP transport, two tools: `search_employee`, `get_leave_balance`). Each gets its own Dev Tunnel to expose it to the internet. They're connected to the agent in four ways: add a REST tool directly from the OpenAPI definition; build a custom connector from the same OpenAPI in Power Apps maker and add it as a tool; call it manually in a topic with the Send HTTP request node and parse the JSON; connect mcp-server as an MCP tool. Finally, switch M1's "leave balance" topic to call one of the tools.

**Tech Stack:** Node.js (built-in `http`, `node:test`, global `fetch`), OpenAPI 3.0, Power Platform custom connector, `@modelcontextprotocol/sdk` (Streamable HTTP), Microsoft Dev Tunnels, Copilot Studio tools.

**Spec:** `docs/superpowers/specs/2026-09-13-ab-620-hr-assistant-design.md` (§4 M3)

- [ ] **Step 4: Write the implementation**

```js
import { createServer } from 'node:http';
import { readFileSync } from 'node:fs';
import { fileURLToPath } from 'node:url';
import { dirname, join } from 'node:path';

const __dirname = dirname(fileURLToPath(import.meta.url));
const API_KEY = process.env.EMPLOYEE_API_KEY || 'dev-local-key';
const FAIL_MODE = process.env.FAIL_MODE || '';

function loadEmployees() {
  const csvPath = join(__dirname, '..', '..', 'data', 'employees.csv');
  const lines = readFileSync(csvPath, 'utf8').trim().split('\n');
  const headers = lines[0].split(',');
  return lines.slice(1).map((line) => {
    const values = line.split(',');
    const record = {};
    headers.forEach((h, i) => { record[h] = values[i]; });
    record.AnnualLeaveBalance = Number(record.AnnualLeaveBalance);
    record.SickLeaveBalance = Number(record.SickLeaveBalance);
    return record;
  });
}

function sendJson(res, status, body) {
  const data = JSON.stringify(body);
  res.writeHead(status, { 'Content-Type': 'application/json' });
  res.end(data);
}

export function createApp() {
  const employees = loadEmployees();

  return createServer((req, res) => {
    const url = new URL(req.url, 'http://localhost');

    if (req.headers['x-api-key'] !== API_KEY) {
      return sendJson(res, 401, { error: 'invalid_api_key' });
    }
    if (FAIL_MODE === '500') {
      return sendJson(res, 500, { error: 'internal_error' });
    }
    if (FAIL_MODE === 'timeout') {
      return;
    }

    const leaveMatch = url.pathname.match(/^\/employees\/([^/]+)\/leave$/);
    const empMatch = url.pathname.match(/^\/employees\/([^/]+)$/);

    if (leaveMatch) {
      const emp = employees.find((e) => e.EmployeeId === leaveMatch[1]);
      if (!emp) return sendJson(res, 404, { error: 'not_found' });
      return sendJson(res, 200, {
        EmployeeId: emp.EmployeeId,
        AnnualLeaveBalance: emp.AnnualLeaveBalance,
        SickLeaveBalance: emp.SickLeaveBalance,
      });
    }

    if (url.pathname === '/employees') {
      const department = url.searchParams.get('department');
      if (!department) return sendJson(res, 400, { error: 'department_required' });
      const matches = employees.filter(
        (e) => e.Department.toLowerCase() === department.toLowerCase()
      );
      return sendJson(res, 200, matches);
    }

    if (empMatch) {
      const emp = employees.find((e) => e.EmployeeId === empMatch[1]);
      if (!emp) return sendJson(res, 404, { error: 'not_found' });
      return sendJson(res, 200, emp);
    }

    sendJson(res, 404, { error: 'not_found' });
  });
}

if (process.argv[1] === fileURLToPath(import.meta.url)) {
  const port = process.env.PORT || 3000;
  createApp().listen(port, () => {
    console.log(`employee-api listening on :${port}`);
  });
}
```

- [ ] **Step 3: Write server.js**

```js
import express from 'express';
import { randomUUID } from 'node:crypto';
import { readFileSync } from 'node:fs';
import { fileURLToPath } from 'node:url';
import { dirname, join } from 'node:path';
import { McpServer } from '@modelcontextprotocol/sdk/server/mcp.js';
import { StreamableHTTPServerTransport } from '@modelcontextprotocol/sdk/server/streamableHttp.js';
import { isInitializeRequest } from '@modelcontextprotocol/sdk/types.js';
import { z } from 'zod';

const __dirname = dirname(fileURLToPath(import.meta.url));

function loadEmployees() {
  const csvPath = join(__dirname, '..', '..', 'data', 'employees.csv');
  const lines = readFileSync(csvPath, 'utf8').trim().split('\n');
  const headers = lines[0].split(',');
  return lines.slice(1).map((line) => {
    const values = line.split(',');
    const record = {};
    headers.forEach((h, i) => { record[h] = values[i]; });
    record.AnnualLeaveBalance = Number(record.AnnualLeaveBalance);
    record.SickLeaveBalance = Number(record.SickLeaveBalance);
    return record;
  });
}

const employees = loadEmployees();

function createMcpServer() {
  const server = new McpServer({ name: 'hr-assistant-mcp', version: '1.0.0' });

  server.registerTool(
    'search_employee',
    {
      description: 'Search employees by a substring of their name or department. Returns EmployeeId, Name and Department for each match.',
      inputSchema: { query: z.string().describe('Part of a name or department to search for') },
    },
    async ({ query }) => {
      const q = query.toLowerCase();
      const matches = employees.filter(
        (e) => e.Name.toLowerCase().includes(q) || e.Department.toLowerCase().includes(q)
      );
      return { content: [{ type: 'text', text: JSON.stringify(matches) }] };
    }
  );

  server.registerTool(
    'get_leave_balance',
    {
      description: 'Get the remaining annual and sick leave days for one employee, looked up by EmployeeId (e.g. E001).',
      inputSchema: { employeeId: z.string().describe('Employee id, e.g. E001') },
    },
    async ({ employeeId }) => {
      const emp = employees.find((e) => e.EmployeeId === employeeId);
      if (!emp) {
        return { content: [{ type: 'text', text: `No employee with id ${employeeId}` }], isError: true };
      }
      return {
        content: [{
          type: 'text',
          text: JSON.stringify({
            EmployeeId: emp.EmployeeId,
            AnnualLeaveBalance: emp.AnnualLeaveBalance,
            SickLeaveBalance: emp.SickLeaveBalance,
          }),
        }],
      };
    }
  );

  return server;
}

const app = express();
app.use(express.json());
const transports = {};

app.post('/mcp', async (req, res) => {
  const sessionId = req.headers['mcp-session-id'];
  let transport = sessionId && transports[sessionId];

  if (!transport) {
    if (sessionId || !isInitializeRequest(req.body)) {
      res.status(400).json({ error: 'invalid_session' });
      return;
    }
    transport = new StreamableHTTPServerTransport({
      sessionIdGenerator: () => randomUUID(),
      onsessioninitialized: (newId) => { transports[newId] = transport; },
    });
    const server = createMcpServer();
    await server.connect(transport);
  }

  await transport.handleRequest(req, res, req.body);
});

app.get('/mcp', async (req, res) => {
  const transport = transports[req.headers['mcp-session-id']];
  if (!transport) {
    res.status(400).send('invalid session');
    return;
  }
  await transport.handleRequest(req, res);
});

const port = process.env.PORT || 3100;
app.listen(port, () => console.log(`mcp-server listening on :${port}`));
```
